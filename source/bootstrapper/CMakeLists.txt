# SVG files remain the editable masters. The matching checked-in PNGs are
# release assets so bootstrapper builds do not depend on an SVG renderer being
# installed (or on renderer-specific filter output).
set(_icon_source_dir "${D_CWD}/assets/icons")
set(_icon_output_dir "${CMAKE_CURRENT_BINARY_DIR}/generated/assets")
file(GLOB _icon_svg_files CONFIGURE_DEPENDS "${_icon_source_dir}/icon_xml_*.svg")

set(_generated_icon_files)
foreach(_svg_file IN LISTS _icon_svg_files)
	get_filename_component(_icon_name "${_svg_file}" NAME_WE)
	set(_raster_source "${D_CWD}/assets/${_icon_name}.png")
	set(_png_file "${_icon_output_dir}/${_icon_name}.png")
	if(NOT EXISTS "${_raster_source}")
		message(FATAL_ERROR "Missing rendered UI asset: ${_raster_source}")
	endif()
	add_custom_command(
		OUTPUT "${_png_file}"
		COMMAND ${CMAKE_COMMAND} -E make_directory "${_icon_output_dir}"
		COMMAND ${CMAKE_COMMAND} -E copy_if_different "${_raster_source}" "${_png_file}"
		DEPENDS "${_raster_source}" "${_svg_file}"
		COMMENT "Staging ${_icon_name}.png"
		VERBATIM
	)
	list(APPEND _generated_icon_files "${_png_file}")
endforeach()

# Generate the icon embedding blocks and the runtime manifest from the icon
# list above. Previously each icon had to be added in three hand-maintained
# places — a 12-line .incbin block in daemon.c, an extern pair in main.cpp and
# a write_blob_file() call — none of which fail the build when forgotten.
set(_icon_asm "${_icon_output_dir}/icon_assets.inc")
set(_icon_manifest "${_icon_output_dir}/icon_manifest.inc")
set(_asm_body "/* Generated by CMake from bootstrapper/assets — do not edit. */\n")
set(_manifest_body "/* Generated by CMake from bootstrapper/assets — do not edit. */\n")
foreach(_icon_file IN LISTS _generated_icon_files)
	get_filename_component(_icon_name "${_icon_file}" NAME_WE)
	string(APPEND _asm_body
		"\".global ${_icon_name}_start\\n\"\n"
		"\".type   ${_icon_name}_start, @object\\n\"\n"
		"\".align  16\\n\"\n"
		"\"${_icon_name}_start:\\n\"\n"
		"\".incbin \\\"\" HYPERHEN_ICON_ASSET_DIR \"/${_icon_name}.png\\\"\\n\"\n"
		"\"${_icon_name}_end:\\n\"\n"
		"\".global ${_icon_name}_size\\n\"\n"
		"\".type   ${_icon_name}_size, @object\\n\"\n"
		"\".align  4\\n\"\n"
		"\"${_icon_name}_size:\\n\"\n"
		"\".int    ${_icon_name}_end - ${_icon_name}_start\\n\"\n")
	string(APPEND _manifest_body "ONION_ICON(${_icon_name})\n")
endforeach()
file(GENERATE OUTPUT "${_icon_asm}" CONTENT "${_asm_body}")
file(GENERATE OUTPUT "${_icon_manifest}" CONTENT "${_manifest_body}")

# Declare the compressed payload as a real build output.  The unpacker embeds
# these files, so a POST_BUILD-only command can otherwise leave it stale until
# a second Ninja invocation.
set(_bootstrapper_lzma "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/bootstrapper.elf.lzma")
set(_bootstrapper_lzma_size "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/bootstrapper.elf.lzma.size")
set(_bootstrapper_sha1 "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/bootstrapper.elf.sha1")
add_custom_command(
	OUTPUT "${_bootstrapper_lzma}" "${_bootstrapper_lzma_size}" "${_bootstrapper_sha1}"
	COMMAND ${CMAKE_COMMAND} -E env "TARGET_ELF=$<TARGET_FILE:${PROJECT_NAME}>"
			${PROJECT_ROOT}/../scripts/pack_bootstrapper.sh
	DEPENDS ${PROJECT_NAME}
	COMMENT "Packing bootstrapper (lzma + size + sha1 of uncompressed ELF)"
	VERBATIM
)
add_custom_target(${PROJECT_NAME}_packed
	DEPENDS "${_bootstrapper_lzma}" "${_bootstrapper_lzma_size}" "${_bootstrapper_sha1}")
add_dependencies(${PROJECT_NAME}_packed ${PROJECT_NAME})
